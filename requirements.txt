# TagScribeR shared dependencies (all platforms / GPU backends).
#
# torch / torchvision are deliberately NOT listed: tools/install.py installs the
# correct build first (ROCm multi-arch, CUDA 12.8, or CPU). Listing torch here
# would let pip replace a ROCm install with a CUDA wheel.
#
# The torch / ROCm stack matches the Fizgig trainer exactly (see tools/install.py).
# Transformers deliberately differs: Fizgig pins 4.57.x for its Qwen3-VL text
# encoder, but every 2026 VLM family (Qwen3.5/3.6/3.8 = qwen3_5, Gemma 4 = gemma4)
# requires Transformers 5.x. Qwen3-VL remains fully supported in 5.x.

# UI
PySide6==6.10.1
qt-material==2.17
qtawesome>=1.3             # Material Design icons (falls back to text if missing)

# Local VLM inference
transformers>=5.18,<6
accelerate>=1.6.0
safetensors>=0.5.3
huggingface-hub
hf_xet>=1.0            # faster downloads from Xet-backed HF repos (optional accelerator)
sentencepiece==0.2.1   # tokenizer backend for several VLM families

# Training (SDXL-family UNet, VAE and CLIP loading; other families ship their own model code)
diffusers>=0.35

# torch.compile (inductor) backend - Krea 2's Compile Blocks speedup needs it (Fizgig requirements.txt). Linux torch
# wheels already depend on triton; Windows needs the community triton-windows build. Also needs the MSVC C++ Build
# Tools on Windows (not pip-installable). If triton is missing or incompatible, compile bails gracefully to eager
# with a console note; training still runs. PINNED to what torch 2.10 pairs with (the Linux torch wheel depends on
# triton 3.6; 3.5.1 is field-proven in Fizgig). Unpinned, a fresh install picked up triton-windows 3.8 (built for a
# newer torch) and previews hung inside torch.compile with no log line. Bump together with torch.
triton-windows>=3.5.1,<3.7 ; sys_platform == 'win32'

# Tagger (WD14 / SmilingWolf ONNX models)
onnxruntime>=1.20

# Remote / OpenAI-compatible APIs (LM Studio, Ollama, llama.cpp server, vLLM, cloud)
openai>=1.50

# Images & metadata
pillow==12.3.0
numpy>=1.26.4,<3
opencv-python==4.10.0.84
piexif==1.1.3

# Safety & platform
send2trash>=1.8        # deletes go to the Recycle Bin
keyring>=25            # API keys in Windows Credential Manager / OS keychain
psutil>=5.9            # RAM checks before loading models; the Train tab's RAM bar
nvidia-ml-py>=12.0     # fast NVIDIA VRAM reads for the Train tab's memory bar (falls back to nvidia-smi)
